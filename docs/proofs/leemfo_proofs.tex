\title{Closed-form envelopes of the Fourier-optics kernel\\
in aberration-corrected LEEM}
\author{Wilson Lau}
\date{25 August 2026}

\begin{document}
\maketitle

\begin{abstract}
We present closed-form spatial and chromatic envelopes of the
Fourier-optics (FO) kernel for image formation in aberration-corrected
low-energy electron microscopy (ac-LEEM), following Yu, Lau and
Altman~\cite{Yu2019}.
That work reduced the partially coherent ensemble integrals to
Gaussian envelopes and compared FO with the contrast-transfer function
(CTF); the closed forms were stated, not derived.
After the first-order tilt, frozen-aperture and mixed tilt--energy
truncations of their Eqs.~(3)--(4), the remaining Gaussian integrals
are identities: they yield the spatial envelope, the chromatic
envelopes including the polar form of the chromatic prefactor, the
FO/CTF ratios, and the Jacobi--Anger expansion of a sinusoidal phase
object.
The formulae are machine-checked in Lean~4 against that kernel and
are the forward map inverted in the companion
reconstruction~\cite{Lau2026inverse}.
This work places FO/CTF disagreement on identities of the same kernel
used to predict a focal series and to recover the object that produced
it.
\end{abstract}

\section{Introduction}
\label{sec:intro}

Image interpretation in LEEM~\cite{Bauer1994,Bauer2014,Tromp2000,Altman2010}
requires a quantitative account of how the cathode lens, contrast
aperture, source size and energy spread map the object exit wave to
the recorded intensity.
Aberration-corrected instruments~\cite{Tromp2010,Tromp2013} make that
map sharper, but they do not remove the distinction between two
standard calculations.
The contrast-transfer function (CTF) evaluates the inverse transforms
of $\boldsymbol{\Psi}H$ and $\boldsymbol{\Psi}^*H^*$ separately and is computationally
efficient~\cite{Schramm2012,Tromp2012}.
Fourier optics (FO) retains inseparable mixing of spatial-frequency
pairs $(\mathbf{q},\mathbf{q}')$ under partial
coherence~\cite{Pang2009,Yu2017,Yu2019}.
The mathematically rigorous FO calculation gains an advantage under
conditions, particularly defocus, in which partial coherence
compromises the validity of the CTF intensity~\cite{Yu2019}.
That distinction is observed for strongly rippled graphene, where CTF
modelling fails and FO reproduces the measured through-focal
contrast~\cite{Yu2017}.

Yu, Lau and Altman~\cite{Yu2019} write the ensemble-averaged kernel as
an integral of a coherent kernel against a Gaussian source density and
a Gaussian energy density, extending Pang~\emph{et al.}~\cite{Pang2009}
to aberration-corrected columns ($C_5$, $C_{CC}$, $C_{3C}$ in place of
$C_3$, $C_C$).
Closed-form envelopes---their Eq.~(5), Eqs.~(6a)--(6b), the polar form
of the chromatic prefactor, and the FO/CTF ratios $\Gamma_S$ and
$\Gamma_C$---are stated in an appendix.
They enter every FO intensity used to interpret a focal series, and
they are the forward kernel of the companion through-focal
inversion~\cite{Lau2026inverse}.
They were not derived.
Here those integrals are evaluated.
After the truncations of paper Eqs.~(3)--(4) they are identities;
the truncations themselves are modelling hypotheses.
Selected identities are machine-checked in Lean~4~\cite{deMoura2021,mathlib2020}
against that kernel (\cref{app:formal}).
Experimental claims and the numerical accuracy of Table~1
in~\cite{Yu2019} are not claimed.

\section{Image formation}
\label{sec:formation}

The object is the complex exit wave
$\psi_0(\mathbf{r})=\sigma(\mathbf{r})\,\ee^{\ii\phi(\mathbf{r})}$,
with $\phi$ in radians.
Spatial frequency is $\mathbf{q}=\boldsymbol{\alpha}/\lambda$, where
$\boldsymbol{\alpha}$ is the emission-angle vector from the virtual object
and $\lambda$ is the wavelength after acceleration to column energy
$E$~\cite{Tromp2012,Schramm2012}.
Vectors are bold and $q=\abs{\mathbf{q}}$; in one dimension $q,q',k$
are signed scalars, so $qq'$ is a product of signed reals.
The Fourier kernel is the unitary $2\pi$-in-the-exponent convention
$\ee^{2\pi\ii\,\mathbf{q}\cdot\mathbf{r}}$, with
$\dd^2q=\dd q_x\,\dd q_y$.
Wave aberration $\chi$ is measured in waves, with transfer factor
$W=\ee^{2\pi\ii\chi}$.
Object phase is independent of that $2\pi$ convention.
The principal branch of the complex logarithm has argument in
$(-\pi,\pi]$, and $z^{1/2}=\exp(\tfrac12\mathrm{Log}\,z)$ with
$z^{1/2}>0$ for $z>0$ real.

With $\boldsymbol{\Psi}=\mathcal{F}[\psi_0]$, the FO intensity is bilinear
in the object spectrum~\cite{Goodman2005,Pang2009,Yu2019},
\begin{equation}
\label{eq:I}
I(\mathbf{r},\Dz)
=\iint
\boldsymbol{\Psi}(\mathbf{q})\,\boldsymbol{\Psi}^*(\mathbf{q}')
\,R(\mathbf{q},\mathbf{q}',\Dz)
\,\ee^{2\pi\ii(\mathbf{q}-\mathbf{q}')\cdot\mathbf{r}}
\,\dd^2q\,\dd^2q'.
\end{equation}
A circular contrast aperture of cutoff $\qap=\alpha_{\mathrm{ap}}/\lambda$
is the indicator $M(\mathbf{q})=\mathbf{1}_{\{\abs{\mathbf{q}}\le\qap\}}$,
which is real.

\begin{approximation}[Frozen aperture]
\label{approx:aperture}
For a small source, $M(\mathbf{q}+\mathbf{k})\approx M(\mathbf{q})$.
This is a modelling hypothesis, not an identity.
\end{approximation}

Spherical aberration and defocus enter as
\begin{equation}
\label{eq:chiS}
\chi_S(\mathbf{q},\Dz)
=\tfrac14 C_3\lambda^3 q^4
+\tfrac16 C_5\lambda^5 q^6
+\tfrac12\Dz\,\lambda\,q^2,
\end{equation}
with $W_S=\ee^{2\pi\ii\chi_S}$.
The sign of $\Dz$ is the Tromp--Schramm convention: $\Dz>0$ adds
positive optical path for off-axis rays~\cite{Tromp2012}.
Mixing the opposite sign used in some simulators interchanges
overfocus and underfocus.
Non-corrected (nac) LEEM keeps $C_3$ and drops $C_5$;
aberration-corrected (ac) LEEM sets $C_3=0$ and keeps $C_5$.
Chromatic aberration is
\begin{equation}
\label{eq:chiC}
\chi_C(\mathbf{q},\varepsilon)
=\tfrac12 C_C\lambda\Bigl(\frac{\varepsilon}{E}\Bigr)q^2
+\tfrac12 C_{CC}\lambda\Bigl(\frac{\varepsilon}{E}\Bigr)^2 q^2
+\tfrac14 C_{3C}\lambda^3\Bigl(\frac{\varepsilon}{E}\Bigr)q^4,
\end{equation}
with $W_C=\ee^{2\pi\ii\chi_C}$.
Nac keeps $C_C$ and drops $C_{CC},C_{3C}$; ac sets $C_C=0$ and keeps
$C_{CC},C_{3C}$.
If $\chi$ is real then
$W(\mathbf{q})W^*(\mathbf{q}')=\ee^{2\pi\ii(\chi(\mathbf{q})-\chi(\mathbf{q}'))}$;
at $\varepsilon=0$, $\chi_C(\cdot,0)=0$ and $W_C(\cdot,0)=1$.

Viewing~\eqref{eq:chiS} as a radial function on $\R^2$,
\begin{equation}
\label{eq:grad}
\nabla_{\mathbf{q}}\chi_S(\mathbf{q},\Dz)
=\bigl(C_3\lambda^3 q^2+C_5\lambda^5 q^4+\Dz\,\lambda\bigr)\,\mathbf{q}.
\end{equation}
In one dimension,
$\partial_q\chi_S=C_3\lambda^3 q^3+C_5\lambda^5 q^5+\Dz\,\lambda\,q$,
so odd powers keep the sign of $q$.

The coherent kernel is paper Eq.~(2)~\cite{Yu2019},
\begin{equation}
\label{eq:R0}
\Rzero(\mathbf{q},\mathbf{q}',\Dz,\varepsilon)
=M(\mathbf{q})M(\mathbf{q}')
\,W_S(\mathbf{q},\Dz)\,W_S^*(\mathbf{q}',\Dz)
\,W_C(\mathbf{q},\varepsilon)\,W_C^*(\mathbf{q}',\varepsilon),
\end{equation}
and the ensemble-averaged kernel is paper Eq.~(1),
\begin{equation}
\label{eq:R}
R(\mathbf{q},\mathbf{q}',\Dz)
=\int_{\R^2}\dd^2k\int_{\R}\dd\varepsilon\;
s(\mathbf{k})\,c(\varepsilon)\,
\Rzero(\mathbf{q}+\mathbf{k},\mathbf{q}'+\mathbf{k},\Dz,\varepsilon).
\end{equation}
The source and energy densities are normalised Gaussians of widths
$\sill>0$ and $\sE>0$,
\begin{equation}
\label{eq:cs}
\begin{aligned}
c(\varepsilon)
&=\frac{1}{\sE\sqrt{2\pi}}
\exp\Bigl(-\frac{\varepsilon^2}{2\sE^2}\Bigr),\\
s_{1\mathrm{D}}(k)
&=\frac{1}{\sill\sqrt{2\pi}}
\exp\Bigl(-\frac{k^2}{2\sill^2}\Bigr),
\qquad
s(\mathbf{k})
=s_{1\mathrm{D}}(k_x)\,s_{1\mathrm{D}}(k_y).
\end{aligned}
\end{equation}
The paper identifies $\DE$ with the FWHM of $c$ and
$\qill=\alpha_{\mathrm{ill}}/\lambda$ with the isotropic FWHM of $s$.
A centred Gaussian $f(x)=\exp(-x^2/(2\sigma^2))$ has
FWHM $\Delta=2\sqrt{2\ln 2}\,\sigma$, hence
$\sigma^2=\Delta^2/(8\ln 2)$ (\cref{lem:fwhm}).

Define the tilt vector
\begin{equation}
\label{eq:a}
\mathbf{a}(\mathbf{q},\mathbf{q}',\Dz)
:=\nabla\chi_S(\mathbf{q},\Dz)-\nabla\chi_S(\mathbf{q}',\Dz).
\end{equation}
On the diagonal $\mathbf{a}(\mathbf{q},\mathbf{q})=0$; on the axis
$\mathbf{q}'=\mathbf{0}$, $\mathbf{a}(\mathbf{q},\mathbf{0})=\nabla\chi_S(\mathbf{q})$.
If $E\neq 0$, the chromatic path difference splits exactly as
$\chi_C(\mathbf{q},\varepsilon)-\chi_C(\mathbf{q}',\varepsilon)
=b_1\varepsilon+b_2\varepsilon^2$, with
\begin{equation}
\label{eq:b1b2}
\begin{aligned}
b_1(\mathbf{q},\mathbf{q}')
&=\frac{C_C\lambda}{2E}(q^2-{q'}^2)
+\frac{C_{3C}\lambda^3}{4E}(q^4-{q'}^4),\\
b_2(\mathbf{q},\mathbf{q}')
&=\frac{C_{CC}\lambda}{2E^2}(q^2-{q'}^2).
\end{aligned}
\end{equation}
This is an identity: no other powers of $\varepsilon$ appear.
On the diagonal $b_1=b_2=0$; both coefficients are antisymmetric in
$(\mathbf{q},\mathbf{q}')$.
Under ac, $C_C=0$ and $b_1=(C_{3C}\lambda^3/(4E))(q^4-{q'}^4)$;
under nac, $C_{CC}=C_{3C}=0$, so $b_2=0$.

\begin{approximation}[First-order tilt, no mixed $k\varepsilon$]
\label{approx:taylor}
Expand the path functions to first order in $\mathbf{k}$ and drop mixed
tilt--energy terms (paper Eq.~(3)):
\begin{equation}
\label{eq:paper3}
\chi_S(\mathbf{q}+\mathbf{k},\Dz)+\chi_C(\mathbf{q}+\mathbf{k},\varepsilon)
\approx
\chi_S(\mathbf{q},\Dz)
+\mathbf{k}\cdot\nabla\chi_S(\mathbf{q},\Dz)
+\chi_C(\mathbf{q},\varepsilon).
\end{equation}
Together with~\cref{approx:aperture} this is paper Eq.~(4)~\cite{Yu2019}:
\begin{equation}
\label{eq:R0approx}
\Rzero(\mathbf{q}+\mathbf{k},\mathbf{q}'+\mathbf{k},\Dz,\varepsilon)
\approx
\Rzero(\mathbf{q},\mathbf{q}',\Dz,0)
\,\ee^{2\pi\ii\,\mathbf{a}\cdot\mathbf{k}}
\,\ee^{2\pi\ii(b_1\varepsilon+b_2\varepsilon^2)}.
\end{equation}
The $(\mathbf{k},\varepsilon)$ integral then factors as
$\EStot\,\ECtot$.
\end{approximation}
Because $\chi_S$ is a polynomial the Taylor remainder is
$O(\abs{\mathbf{k}}^2)$; \eqref{eq:paper3} does not justify dropping
that remainder uniformly in $\mathbf{q}$.

Under~\cref{approx:taylor} the working kernel is
\begin{equation}
\label{eq:assembly}
R(\mathbf{q},\mathbf{q}',\Dz)
=\Rzero(\mathbf{q},\mathbf{q}',\Dz,0)\,
\EStot(\mathbf{q},\mathbf{q}',\Dz)\,
\ECtot(\mathbf{q},\mathbf{q}'),
\end{equation}
with $\EStot$ from~\eqref{eq:paper5} and $\ECtot$ from~\eqref{eq:paper6a}.
Nac/ac specialisation is substitution of coefficients, not a new
identity.

\section{Spatial envelope}
\label{sec:spatial}

After~\cref{approx:taylor} the spatial envelope is
\begin{equation}
\label{eq:ESint}
\EStot(\mathbf{q},\mathbf{q}',\Dz)
:=\int_{\R^2}
s(\mathbf{k})\,\ee^{2\pi\ii\,\mathbf{a}\cdot\mathbf{k}}\,\dd^2k.
\end{equation}
The characteristic function of a centred Gaussian of width $\sigma$ is
\begin{equation}
\label{eq:char1d}
\int_{\R}g_\sigma(k)\,\ee^{2\pi\ii a k}\,\dd k
=\exp(-2\pi^2\sigma^2 a^2),
\end{equation}
and the two-dimensional source factors, so
\begin{equation}
\label{eq:char2d}
\int_{\R^2}s(\mathbf{k})\,\ee^{2\pi\ii\,\mathbf{a}\cdot\mathbf{k}}\,\dd^2k
=\exp(-2\pi^2\sill^2\abs{\mathbf{a}}^2)
\end{equation}
(\cref{lem:cgaussian,thm:char}).
If $\qill>0$ and $\kappa=\pi^2\qill^2/(4\ln 2)$, the FWHM conversion
$\sill^2=\qill^2/(8\ln 2)$ rewrites~\eqref{eq:char2d} as paper Eq.~(5),
\begin{equation}
\label{eq:paper5}
\EStot(\mathbf{q},\mathbf{q}',\Dz)
=\exp\Bigl(-\frac{\pi^2\qill^2}{4\ln 2}\,
\abs{\mathbf{a}(\mathbf{q},\mathbf{q}',\Dz)}^2\Bigr)
=\ee^{-\kappa\abs{\mathbf{a}}^2}.
\end{equation}
The envelope is real, even under
$(\mathbf{q},\mathbf{q}')\leftrightarrow(\mathbf{q}',\mathbf{q})$,
and equals $1$ on the diagonal.
Thus $\kappa=2\pi^2\sill^2$, not $\pi^2\sigma^2$ (a factor of two);
substituting FWHM for $\sigma$ instead of $\sigma^2=\Delta^2/(8\ln 2)$
is a factor $8\ln 2$.

Writing $u(\mathbf{q}):=C_3\lambda^3 q^2+C_5\lambda^5 q^4+\Dz\lambda$,
one has $\nabla\chi_S=u(\mathbf{q})\,\mathbf{q}$ and
$\mathbf{a}=u(\mathbf{q})\mathbf{q}-u(\mathbf{q}')\mathbf{q}'$, hence
$\abs{\mathbf{a}}^2
=u(\mathbf{q})^2 q^2+u(\mathbf{q}')^2{q'}^2
-2\,u(\mathbf{q})u(\mathbf{q}')\,(\mathbf{q}\cdot\mathbf{q}')$.
The inner product $\mathbf{q}\cdot\mathbf{q}'$ cannot be replaced by
$q\,q'$.
In one dimension~\eqref{eq:paper5} is
\begin{equation}
\label{eq:ES1d}
\EStot(q,q',\Dz)
=\exp\Biggl(
-\kappa
\Bigl[
C_3\lambda^3(q^3-{q'}^3)
+C_5\lambda^5(q^5-{q'}^5)
+\Dz\lambda(q-q')
\Bigr]^2
\Biggr);
\end{equation}
one may factor $(q-q')^2$ via $q^3-{q'}^3=(q-q')(q^2+qq'+{q'}^2)$.
For ac, $C_3=0$.
The CTF slice is
$\EStot(\mathbf{q},\mathbf{0},\Dz)=\exp(-\kappa\abs{\nabla\chi_S(\mathbf{q})}^2)$.

\section{Chromatic envelope}
\label{sec:chromatic}

The chromatic envelope is
\begin{equation}
\label{eq:ECint}
\ECtot(\mathbf{q},\mathbf{q}')
:=\int_{\R}
c(\varepsilon)\,
\exp\bigl(2\pi\ii(b_1\varepsilon+b_2\varepsilon^2)\bigr)
\,\dd\varepsilon.
\end{equation}
The integrand is a constant times $\exp(B\varepsilon^2+C\varepsilon)$
with $B=-1/(2\sE^2)+2\pi\ii b_2$ and $C=2\pi\ii b_1$.
If $\sE>0$ then $\ReOp(B)=-1/(2\sE^2)<0$, independently of $b_2$:
Gaussian decay comes only from the real energy density.
Writing $z:=1-4\pi\ii b_2\sE^2$ one has $\ReOp(z)=1>0$ and
$B=-z/(2\sE^2)$.
Evaluating~\cref{lem:cgaussian} on the principal branch
(\cref{thm:EC}) produces paper Eqs.~(6a)--(6b),
\begin{equation}
\label{eq:paper6b}
\ECC(\mathbf{q},\mathbf{q}')
:=\bigl(1-4\pi\ii\,b_2\sE^2\bigr)^{-1/2}
=z^{-1/2},
\end{equation}
with value $1$ at $b_2=0$, and
\begin{equation}
\label{eq:paper6a}
\ECtot
=\ECC\exp\bigl(-2\pi^2\sE^2 b_1^2\,\ECC^2\bigr)
=\bigl(1-4\pi\ii b_2\sE^2\bigr)^{-1/2}
\exp\Bigl(
-\frac{2\pi^2\sE^2 b_1^2}{1-4\pi\ii b_2\sE^2}
\Bigr).
\end{equation}
In FWHM variables,
$z=1-\ii\,(\pi b_2\DE^2)/(2\ln 2)$.
The Han{\ss}en--Trepte packaging~\cite{Hanen1971} writes
$2\pi^2\sE^2 b_1^2=(\DE)^2(2\pi b_1)^2/(16\ln 2)$, where $2\pi b_1$
is the coefficient of $\varepsilon$ in the radian phase
$2\pi\Delta\chi_C$: both $\pi$ (from $W=\ee^{2\pi\ii\chi}$) and
$16\ln 2$ (from FWHM) must appear.

If $b_2=0$ then $\ECC=1$ and $\ECtot=\exp(-2\pi^2\sE^2 b_1^2)$.
On the nac CTF slice $\mathbf{q}'=\mathbf{0}$,
$b_1=C_C\lambda q^2/(2E)$, hence
\begin{equation}
\label{eq:nac-ctf}
\ECtot(\mathbf{q},\mathbf{0})
=\exp\Biggl(
-\frac{\bigl(\pi C_C\lambda\DE\,q^2\bigr)^2}{16\ln 2\,E^2}
\Biggr),
\end{equation}
the standard chromatic envelope of Schramm \emph{et al.}~\cite{Schramm2012}.
The density $c$ is real, so
$\ECtot(\mathbf{q}',\mathbf{q})=\conj{\ECtot(\mathbf{q},\mathbf{q}')}$.

\subsection{Polar form}
\label{sec:polar}

Write $y:=4\pi b_2\sE^2=\pi b_2\DE^2/(2\ln 2)$, so
$\ECC=(1-\ii y)^{-1/2}$.
For $y\in\R$, $\abs{1-\ii y}=\sqrt{1+y^2}$ and
$\Arg(1-\ii y)=-\arctan y$: the real part is identically $1$, so the
argument never jumps.
On the principal branch,
\begin{equation}
\label{eq:ECC-polar}
(1-\ii y)^{-1/2}
=(1+y^2)^{-1/4}
\exp\bigl(\tfrac{\ii}{2}\arctan y\bigr).
\end{equation}
The phase of $\ECC$ therefore lies in $(-\pi/4,\pi/4)$.
Under ac, $y=\gamma(q^2-{q'}^2)$ with
\begin{equation}
\label{eq:gamma}
\gamma:=\frac{\pi C_{CC}\lambda\DE^2}{4\ln 2\,E^2},
\end{equation}
and~\eqref{eq:ECC-polar} is
$\ECC
=\bigl[1+\gamma^2(q^2-{q'}^2)^2\bigr]^{-1/4}
\exp\bigl(\tfrac{\ii}{2}\arctan(\gamma(q^2-{q'}^2))\bigr)$.

Let
\begin{equation}
\label{eq:eta}
\eta:=\frac{(\pi C_{3C}\lambda^3\DE)^2}{64\ln 2\,E^2},
\end{equation}
so that $2\pi^2\sE^2 b_1^2=\eta(q^4-{q'}^4)^2$ under ac.
Rationalising $1/(1-\ii y)=(1+\ii y)/(1+y^2)$ splits the remaining
exponential:
\begin{equation}
\label{eq:exp-polar}
\begin{aligned}
\exp(-2\pi^2\sE^2 b_1^2\,\ECC^2)
&=\exp\Biggl(
-\frac{\eta(q^4-{q'}^4)^2}{1+\gamma^2(q^2-{q'}^2)^2}
\Biggr)\\
&\qquad\cdot
\exp\Biggl(
-\ii\,
\frac{\eta\gamma(q^2-{q'}^2)(q^4-{q'}^4)^2}{1+\gamma^2(q^2-{q'}^2)^2}
\Biggr).
\end{aligned}
\end{equation}
Writing $\ECtot=A\ee^{\ii\theta}$ with $A\ge 0$, moduli multiply and
arguments add:
\begin{equation}
\label{eq:A-theta}
\begin{aligned}
A(q,q')
&=\bigl[1+\gamma^2(q^2-{q'}^2)^2\bigr]^{-1/4}
\exp\Biggl(
-\frac{\eta(q^4-{q'}^4)^2}{1+\gamma^2(q^2-{q'}^2)^2}
\Biggr),\\
\theta(q,q')
&=\tfrac12\arctan\bigl(\gamma(q^2-{q'}^2)\bigr)
-\frac{\eta\gamma(q^2-{q'}^2)(q^4-{q'}^4)^2}{1+\gamma^2(q^2-{q'}^2)^2}.
\end{aligned}
\end{equation}
Parity: $A(q,q')=A(q',q)=A(\abs{q},\abs{q'})$ and
$\theta(q,q')=-\theta(q',q)$.

\section{Comparison with the contrast-transfer function}
\label{sec:ratios}

The paper compares FO with a separable CTF kernel through the
ratios~\cite{Yu2019}
\begin{equation}
\label{eq:paper7}
\begin{aligned}
\Gamma_C(\mathbf{q},\mathbf{q}')
&=\frac{\ECtot(\mathbf{q},\mathbf{0})\,\ECtot(\mathbf{q}',\mathbf{0})}
{\ECtot(\mathbf{q},\mathbf{q}')},
\\
\Gamma_S(\mathbf{q},\mathbf{q}',\Dz)
&=\frac{\EStot(\mathbf{q},\mathbf{0},\Dz)\,\EStot(\mathbf{q}',\mathbf{0},\Dz)}
{\EStot(\mathbf{q},\mathbf{q}',\Dz)}.
\end{aligned}
\end{equation}
Printed~(7a) uses the non-conjugated product
$\ECtot(\mathbf{q},\mathbf{0})\ECtot(\mathbf{q}',\mathbf{0})$.
For ac, $\ECtot$ is complex, so $\Gamma_C$ is complex; the plotted
quantity is $\abs{\Gamma_C}$.
In the polar form of~\cref{sec:polar},
\begin{equation}
\label{eq:gammaC-polar}
\Gamma_C(q,q')
=\frac{A(q,0)\,A(q',0)}{A(q,q')}
\exp\bigl(\ii[\theta(q,0)+\theta(q',0)-\theta(q,q')]\bigr).
\end{equation}

Let $\mathbf{u}=\nabla\chi_S(\mathbf{q},\Dz)$ and
$\mathbf{v}=\nabla\chi_S(\mathbf{q}',\Dz)$.
Then $\EStot=\ee^{-\kappa\abs{\mathbf{u}-\mathbf{v}}^2}$ and the
quadratic terms cancel, leaving
\begin{equation}
\label{eq:gammaS}
\Gamma_S=\exp(-2\kappa\,\mathbf{u}\cdot\mathbf{v}).
\end{equation}
The inner product is $\nabla\chi_S(\mathbf{q})\cdot\nabla\chi_S(\mathbf{q}')$,
not a product of Euclidean norms.
In one dimension with $C_3=0$,
\begin{equation}
\label{eq:gammaS-ac}
\Gamma_S(q,q',\Dz)
=\exp\Biggl[
-\frac{\pi^2\qill^2}{2\ln 2}
\bigl(C_5\lambda^5 q^5+\Dz\lambda\,q\bigr)
\bigl(C_5\lambda^5 {q'}^5+\Dz\lambda\,q'\bigr)
\Biggr].
\end{equation}

If $\qill=\DE=0$ then every envelope equals $1$, hence
$\Gamma_C=\Gamma_S=1$.
On either spatial axis $\Gamma_S=1$.
The chromatic ratio satisfies $\Gamma_C(\mathbf{q},\mathbf{0})=1$;
if $\DE>0$ then only $\abs{\Gamma_C(\mathbf{0},\mathbf{q}')}=1$ in
general for ac, because~\eqref{eq:paper7} has no extra conjugate.

After~\cref{approx:taylor} the working kernels are
\begin{align}
\RFO(\mathbf{q},\mathbf{q}',\Dz)
&:=\Rzero(\mathbf{q},\mathbf{q}',\Dz,0)\,
\EStot(\mathbf{q},\mathbf{q}',\Dz)\,
\ECtot(\mathbf{q},\mathbf{q}'),
\label{eq:RFO}\\
\RCTF(\mathbf{q},\mathbf{q}',\Dz)
&:=\Rzero(\mathbf{q},\mathbf{q}',\Dz,0)\,
\EStot(\mathbf{q},\mathbf{0},\Dz)\,
\EStot(\mathbf{q}',\mathbf{0},\Dz)\,
\ECtot(\mathbf{q},\mathbf{0})\,
\ECtot(\mathbf{q}',\mathbf{0}),
\label{eq:RCTF}
\end{align}
where the last two chromatic factors are not conjugated, matching
printed~(7a).
Neither envelope vanishes ($\EStot$ is an exponential;
$\ECC=z^{-1/2}$ with $\ReOp(z)=1$), so multiplying~\eqref{eq:RFO} by
\eqref{eq:paper7} yields the comparison identity
\begin{equation}
\label{eq:R-gamma}
\RCTF=\RFO\,\Gamma_C\,\Gamma_S.
\end{equation}
On $\mathbf{q}'=\mathbf{0}$ one has $\Gamma_C=\Gamma_S=1$, hence
$\RFO(\mathbf{q},\mathbf{0},\Dz)=\RCTF(\mathbf{q},\mathbf{0},\Dz)$.
Thus $\Gamma_C\Gamma_S\neq 1$ is the warning that a separable CTF
kernel differs from the FO kernel; the two agree on the axes, at
perfect coherence, and in the weak-phase limit that samples those
axes.
The coherent kernel is Hermitian,
$\conj{\Rzero(\mathbf{q},\mathbf{q}',\Dz,\varepsilon)}
=\Rzero(\mathbf{q}',\mathbf{q},\Dz,\varepsilon)$,
because $M$ is real and each of $W_S$ and $W_C$ is unimodular.

\section{Sinusoidal phase object}
\label{sec:jacobi}

A kinematic LEEM phase object of physical amplitude $A$ and wavelength
$\Lambda$ is
$\psi_0(x)\propto\exp(\ii\varphi\sin(kx))$ with
$\varphi=4\pi A/\lambda_0$ and $k=2\pi/\Lambda$~\cite{Yu2017,Yu2019}.
The phase amplitude is a definition of the reflection kinematics, not
an FO theorem.
At $E_0=10\,\mathrm{eV}$, $\lambda_0\simeq 0.39\,\mathrm{nm}$, so
$A=0.1\,\mathrm{nm}$ already gives $\varphi\sim 3$, outside the
weak-phase regime occupied by graphene ripples and atomic
steps~\cite{Yu2017}.
The Bessel coefficients are the Fourier coefficients
\begin{equation}
\label{eq:Jn}
J_n(\varphi)
=\frac{1}{2\pi}
\int_{(0,2\pi]}
\exp(\ii\varphi\sin\theta)\,\ee^{-\ii n\theta}\,\dd\theta,
\end{equation}
and Jacobi--Anger supplies the expansion
\begin{equation}
\label{eq:jacobi}
\exp(\ii\varphi\sin\theta)
=\sum_{n\in\Z}J_n(\varphi)\,\ee^{\ii n\theta},
\end{equation}
absolutely and uniformly on $\R$ (\cref{thm:jacobi}), hence
$\exp(\ii\varphi\sin(kx))=\sum_n J_n(\varphi)\,\ee^{\ii nkx}$.
In the paper's $2\pi$ Fourier convention,
$\ee^{\ii n k x}=\ee^{2\pi\ii q_n x}$ with $q_n=n/\Lambda$.
Larger $\varphi$ populates larger $\abs{n}$, so pairs $(q,q')$ leave
the axes where $\Gamma_C\Gamma_S=1$, and a separable CTF becomes
inaccurate.
That is the object used in~\cite{Yu2017,Yu2019} to exhibit CTF
failure, and the aperture-truncated Bessel ladder inverted in the
companion reconstruction~\cite{Lau2026inverse}.

\section{Conclusions}
\label{sec:end}

We evaluate the closed-form spatial and chromatic envelopes of the FO
kernel of Yu, Lau and Altman~\cite{Yu2019}.
After the truncations of their Eqs.~(3)--(4) those envelopes are
identities: the spatial envelope is the characteristic function of a
Gaussian source, and the chromatic envelopes follow from a complex
Gaussian integral, including the polar form of the chromatic
prefactor.
The comparison $\RCTF=\RFO\,\Gamma_C\Gamma_S$ isolates the off-axis
pairs at which FO and CTF disagree; the two agree on the axes, at
perfect coherence, and for weak-phase objects that sample those axes.
Jacobi--Anger places a sinusoidal object on the frequency ladder where
that distinction matters.
These identities are the forward kernel inverted in the companion
through-focal reconstruction, with selected claims machine-checked in
Lean~4 against that kernel.
FO/CTF disagreement is therefore an identity of the same kernel used
to predict a focal series and to recover the object that produced it.

\appendix
\section{Formal claims}
\label{app:formal}

The statements below are the analytic identities used in the main
text.
\Cref{approx:aperture,approx:taylor} are not theorems: the working
kernel $\RFO$ is defined as the factorised expression after those
truncations.
Wavelength is the field \texttt{lam} (Lean reserves $\lambda$).
No experimental claim or Table~1 numerical accuracy is asserted.

\begin{lemma}[Complex Gaussian integral]
\label{lem:cgaussian}
Let $B,C,D\in\C$ with $\ReOp(B)<0$. Then
\begin{equation}
\label{eq:cgaussian}
\int_{\R}\exp(Bx^2+Cx+D)\,\dd x
=\Bigl(\frac{\pi}{-B}\Bigr)^{1/2}
\exp\Bigl(D-\frac{C^2}{4B}\Bigr),
\end{equation}
using the principal square root.
\end{lemma}
\begin{proof}
Complete the square:
$Bx^2+Cx+D=B\bigl(x+C/(2B)\bigr)^2+D-C^2/(4B)$.
The hypothesis $\ReOp(B)<0$ gives Gaussian decay in a neighbourhood of
the real axis, so the contour may be shifted.
Because $\ReOp(-B)>0$, one has $\Arg(-B)\in(-\pi/2,\pi/2)$ and the
principal square root never meets its branch cut.
When $B$ is real and negative this recovers
$\sqrt{\pi/\abs{B}}$.
\end{proof}

\begin{lemma}[FWHM of a centred Gaussian]
\label{lem:fwhm}
Let $f(x)=\exp(-x^2/(2\sigma^2))$ with $\sigma>0$.
If $f(\pm\Delta/2)=\tfrac12 f(0)$, then
$\Delta=2\sqrt{2\ln 2}\,\sigma$ and
$\sigma^2=\Delta^2/(8\ln 2)$.
\end{lemma}
\begin{proof}
$f(0)=1$, so $\exp\bigl(-(\Delta/2)^2/(2\sigma^2)\bigr)=1/2$.
Taking $-\log$ yields $\Delta^2/(8\sigma^2)=\ln 2$.
\end{proof}

\begin{theorem}[Characteristic functions]
\label{thm:char}
For $\sigma>0$ and $a\in\R$,
\eqref{eq:char1d} holds; for $\mathbf{a}\in\R^2$,
\eqref{eq:char2d} holds with $s$ as in~\eqref{eq:cs}.
\end{theorem}
\begin{proof}
The one-dimensional integrand is a constant times $\exp(Bk^2+Ck)$ with
$B=-1/(2\sigma^2)$ and $C=2\pi\ii a$.
\Cref{lem:cgaussian} cancels the Gaussian normalisation (positive
square root, $\sigma>0$) and produces the exponent $-C^2/(4B)$.
Then $C^2=-4\pi^2 a^2$, so $-C^2/(4B)=-2\pi^2\sigma^2 a^2$.
The two-dimensional source is a product of one-dimensional Gaussians
of the same width, and $\mathbf{a}\cdot\mathbf{k}$ factors the
exponential, so the exponents add.
\end{proof}

\begin{lemma}[Square-root identity]
\label{lem:sqrt-div}
If $A>0$ is real and $D\in\C\setminus\{0\}$ has $\Arg(D)\neq\pi$, then
$(A/D)^{1/2}=A^{1/2}D^{-1/2}$ on principal branches.
This applies to $z=1-4\pi\ii b_2\sE^2$, because $\ReOp(z)=1$.
\end{lemma}
The identity $\mathrm{Log}(1/w)=-\mathrm{Log}\,w$ fails on the
negative reals; $\ReOp(z)=1$ keeps the cut away, so $b_2=0$ yields
$+1$.

\begin{theorem}[Paper Eqs.~(6a)--(6b)]
\label{thm:EC}
If $\sE>0$, then~\eqref{eq:paper6b}--\eqref{eq:paper6a} hold on the
principal branch.
\end{theorem}
\begin{proof}
\Cref{lem:cgaussian} produces the prefactor
$(1/(\sE\sqrt{2\pi}))(\pi/(-B))^{1/2}$.
Then $\pi/(-B)=2\pi\sE^2/z$.
\Cref{lem:sqrt-div} cancels the Gaussian normalisation, leaving
$z^{-1/2}=\ECC$.
At $b_2=0$, $z=1$ and $1^{-1/2}=1$ (not $-1$).
The exponent is $-C^2/(4B)$ with $C=2\pi\ii b_1$, hence
$C^2=-4\pi^2 b_1^2$ and $-C^2/(4B)=-2\pi^2\sE^2 b_1^2/z$.
Forgetting $\ii^2=-1$ produces a purely oscillatory ``envelope''.
Since $\ECC^2=z^{-1}$ this is $-2\pi^2\sE^2 b_1^2\,\ECC^2$.
\end{proof}

\begin{theorem}[Jacobi--Anger]
\label{thm:jacobi}
For all real $\varphi,\theta$, \eqref{eq:jacobi} converges absolutely
and uniformly on $\R$.
\end{theorem}
\begin{proof}
The map $\theta\mapsto\exp(\ii\varphi\sin\theta)$ is continuous,
$2\pi$-periodic and unimodular; its second derivative is bounded by
$\abs{\varphi}+\varphi^2$.
Two integrations by parts on the circle bound
$\abs{J_n(\varphi)}\le(\abs{\varphi}+\varphi^2)/n^2$ for $n\neq 0$,
so the coefficients are absolutely summable.
A continuous function on the circle with absolutely summable Fourier
coefficients equals its Fourier series pointwise; both sides of
\eqref{eq:jacobi} are $2\pi$-periodic, so the identity extends to all
real $\theta$.
\end{proof}
